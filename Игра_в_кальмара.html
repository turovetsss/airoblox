Красный свет / зелёный свет — арена, кукла, таймер, урон за движение

🚦
Миссия проекта: собрать арену и режим «Красный свет — зелёный свет». На зелёном — беги к финишу. На красном — стой: сервер ловит движение и снимает HP. Кукла поворачивает голову, GUI показывает свет, здоровье и таймер.

🚦 Два состояния

Зелёный свет

Можно бежать. Кукла «смотрит» на финиш. Урон за движение не наносится.

Красный свет

Стоим! Сервер сравнивает позицию каждые 0.2 с. Сдвинулся > 0.5 → −20 HP.

Внутри GameGUI создай LocalScript и вставь код обновления UI.

📄 LocalScript (GameGUI)
local player = game.Players.LocalPlayer
local replicatedStorage = game:GetService("ReplicatedStorage")
local lightStateEvent = replicatedStorage:WaitForChild("LightStateEvent")
local timerUpdateEvent = replicatedStorage:WaitForChild("TimerUpdateEvent")
local screenGui = script.Parent
local lightLabel = screenGui:WaitForChild("LightLabel")
local timerLabel = screenGui:WaitForChild("TimerLabel")
local healthLabel = screenGui:WaitForChild("HealthLabel")

local function updateHealth()
  while true do
    local character = player.Character
    if character then
      local humanoid = character:FindFirstChild("Humanoid")
      if humanoid then
        healthLabel.Text = " " .. math.floor(humanoid.Health)
      end
    end
    wait(0.2)
  end
end
spawn(updateHealth)

lightStateEvent.OnClientEvent:Connect(function(isRed, special, message)
  if special == "message" then
    lightLabel.Text = message
    wait(2)
    lightLabel.Text = ""
  elseif special == "timeout" then
    lightLabel.Text = "ВРЕМЯ ВЫШЛО!"
    lightLabel.TextColor3 = Color3.new(1, 0, 0)
  else
    if isRed then
      lightLabel.Text = " КРАСНЫЙ СВЕТ"
      lightLabel.TextColor3 = Color3.new(1, 0, 0)
    else
      lightLabel.Text = " ЗЕЛЁНЫЙ СВЕТ"
      lightLabel.TextColor3 = Color3.new(0, 1, 0)
    end
  end
end)

timerUpdateEvent.OnClientEvent:Connect(function(currentTime)
  timerLabel.Text = " " .. currentTime .. " сек"
end)

Обычный Script в ServerScriptService.

📄 Главный серверный Script (полный код)
local replicatedStorage = game:GetService("ReplicatedStorage")
local players = game:GetService("Players")
local TweenService = game:GetService("TweenService")

local lightStateEvent = replicatedStorage:FindFirstChild("LightStateEvent")
if not lightStateEvent then
  lightStateEvent = Instance.new("RemoteEvent")
  lightStateEvent.Name = "LightStateEvent"
  lightStateEvent.Parent = replicatedStorage
end
local timerUpdateEvent = replicatedStorage:FindFirstChild("TimerUpdateEvent")
if not timerUpdateEvent then
  timerUpdateEvent = Instance.new("RemoteEvent")
  timerUpdateEvent.Name = "TimerUpdateEvent"
  timerUpdateEvent.Parent = replicatedStorage
end

local START_DELAY = 5
local GAME_TIME = 60
local RED_MIN, RED_MAX = 3, 6
local GREEN_MIN, GREEN_MAX = 3, 6
local DAMAGE_PER_MOVE = 20

local gameActive = false
local timer = GAME_TIME
local playersFinished = {}
local playersPositions = {}

local startPlate = workspace:FindFirstChild("StartPlate")
local finishPlate = workspace:FindFirstChild("FinishPlate")
local doll = workspace:FindFirstChild("Doll")
local dollhead = doll and doll:FindFirstChild("Model")
local head = dollhead and dollhead:FindFirstChild("Head")
if not head then warn("Голова куклы не найдена") end

local function rotateHeadTo(target)
  if not head then return end
  local targetPos = target:IsA("BasePart") and target.Position or target
  local direction = (targetPos - head.Position).Unit
  local targetCF = CFrame.lookAt(head.Position, head.Position + direction)
  TweenService:Create(head, TweenInfo.new(0.5), {CFrame = targetCF}):Play()
end

local function setLightState(isRed)
  lightStateEvent:FireAllClients(isRed)
  if isRed then
    if startPlate then rotateHeadTo(startPlate) end
  else
    if finishPlate then rotateHeadTo(finishPlate) end
  end
end

local function damagePlayer(player, amount)
  local character = player.Character
  if not character then return end
  local humanoid = character:FindFirstChild("Humanoid")
  if humanoid and humanoid.Health > 0 then
    humanoid.Health = math.max(0, humanoid.Health - amount)
  end
end

local function checkMovement()
  for _, player in ipairs(players:GetPlayers()) do
    if playersFinished[player] then continue end
    local character = player.Character
    if not character then continue end
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    if rootPart then
      local currentPos = rootPart.Position
      local prevPos = playersPositions[player]
      if prevPos and (currentPos - prevPos).Magnitude > 0.5 then
        damagePlayer(player, DAMAGE_PER_MOVE)
      end
      playersPositions[player] = currentPos
    end
  end
end

local function redLightCycle(duration)
  setLightState(true)
  playersPositions = {}
  for _, player in ipairs(players:GetPlayers()) do
    if not playersFinished[player] and player.Character then
      local rootPart = player.Character:FindFirstChild("HumanoidRootPart")
      if rootPart then playersPositions[player] = rootPart.Position end
    end
  end
  local startTime = tick()
  while gameActive and (tick() - startTime) < duration do
    checkMovement()
    wait(0.2)
  end
end

local function greenLightCycle(duration)
  setLightState(false)
  wait(duration)
end

local function startGame()
  gameActive = true
  timer = GAME_TIME
  playersFinished = {}
  spawn(function()
    while gameActive and timer > 0 do
      timerUpdateEvent:FireAllClients(timer)
      wait(1)
      timer = timer - 1
    end
    if timer <= 0 and gameActive then
      gameActive = false
      lightStateEvent:FireAllClients(nil, "timeout")
    end
  end)
  while gameActive and timer > 0 do
    redLightCycle(math.random(RED_MIN, RED_MAX))
    if not gameActive then break end
    greenLightCycle(math.random(GREEN_MIN, GREEN_MAX))
  end
end

if finishPlate then
  finishPlate.Touched:Connect(function(hit)
    local player = players:GetPlayerFromCharacter(hit.Parent)
    if player and not playersFinished[player] then
      playersFinished[player] = true
      lightStateEvent:FireAllClients(nil, "message", player.Name .. " финишировал!")
      local character = hit.Parent
      if character and character.PrimaryPart then
        character:SetPrimaryPartCFrame(CFrame.new(0, 100, 0))
      end
    end
  end)
end

wait(START_DELAY)
startGame()
🔑
После Play ждём 5 сек (START_DELAY) — затем начинается цикл красный ↔ зелёный.

🩸 Фаза 7 — Эффект крови 10 мин

  1. StarterPlayer → StarterPlayerScripts → LocalScript.
  2. Вставь скрипт частиц при потере HP.

LocalScript в StarterPlayerScripts.

Пример кода эффекта в редакторе Studio.

Логика: HealthChanged → если HP упал → ParticleEmitter на торсе → удалить через 0.5 с.

local player = game.Players.LocalPlayer
local character = player.Character or player.CharacterAdded:Wait()
local humanoid = character:WaitForChild("Humanoid")
local lastHealth = humanoid.Health

humanoid.HealthChanged:Connect(function(health)
  if health < lastHealth then
    local torso = character:FindFirstChild("UpperTorso") or character:FindFirstChild("Torso")
    if torso then
      local blood = Instance.new("ParticleEmitter")
      blood.Texture = "rbxasset://textures/particles/smoke_main.dds"
      blood.Color = ColorSequence.new(Color3.new(1, 0, 0))
      blood.Rate = 50
      blood.Lifetime = NumberRange.new(0.3)
      blood.SpreadAngle = Vector2.new(180, 180)
      blood.Speed = NumberRange.new(2, 5)
      blood.Enabled = true
      blood.Parent = torso
      wait(0.5)
      blood:Destroy()
    end
  end
  lastHealth = health
end)
📌
Сравниваем с lastHealth, чтобы эффект был именно при уроне, а не при любом HealthChanged.
