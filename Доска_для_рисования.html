🎨 Доска для рисования

Пиксели · палитра · тема раунда · таймер · очистка доски

🖌️
Миссия проекта: сделать пиксельную доску, как в «угадай рисунок». Сервер выдаёт тему и таймер, банки краски выбирают цвет, мышь красит «пиксели» на доске. В конце раунда доска вспыхивает и очищается.

💡 На этом проекте ты:

  • сгенерируешь сетку пикселей скриптом (двойной for) 🧱
  • сделаешь палитру с ClickDetector и RemoteEvent 🎨
  • запустишь цикл тем и таймера в GameManager ⏱️
  • покажешь тему и время в GUI, покрасишь пиксели мышью ✏️

Раунд

🔁 Двойной цикл = сетка

Один for по X, второй по Z — получаем квадрат из клеток. При boardSize = 50 будет 2500 пикселей. Studio может чуть «подумать» при Play — это нормально.

for x = 0, boardSize-1 do
  for z = 0, boardSize-1 do
    -- создать Pixel
  end
end

🧱 Фаза 1 — DrawingBoard и пиксели 12 мин

  1. Создай Part, назови DrawingBoard. Сделай широкую плоскую «доску», Anchored = true.
  2. Внутрь добавь Script и вставь код генерации пикселей.
  3. Нажми Play один раз — на доске должна появиться белая сетка. Stop.

Базовая доска — большой плоский Part.

DrawingBoard → Script.

⚠️
Пиксели создаются при каждом запуске скрипта. Если запускал Play несколько раз и на доске «двойной слой» — удали лишние Pixel в Explorer (или очисти дочерние Parts доски) и запусти снова.
📄 Код Script в DrawingBoard
local board = script.Parent
local boardSize = 50 -- пикселей по стороне
local pixelSize = 1
local startX = -boardSize/2 + 0.5
local startZ = -boardSize/2 + 0.5

for x = 0, boardSize-1 do
  for z = 0, boardSize-1 do
    local pixel = Instance.new("Part")
    pixel.Name = "Pixel"
    pixel.Size = Vector3.new(0.9, 0.2, 0.9)
    pixel.Position = board.Position + Vector3.new(startX + x, 0.2, startZ + z)
    pixel.Anchored = true
    pixel.BrickColor = BrickColor.White()
    pixel.Material = Enum.Material.SmoothPlastic
    pixel.Parent = board
  end
end

📄 GameManager
local Players = game:GetService("Players")
local replicatedStorage = game:GetService("ReplicatedStorage")
local gameInfoEvent = replicatedStorage:FindFirstChild("GameInfoEvent")

local themes = {
  " Котик",
  " Собачка",
  " Домик",
  " Солнышко",
  " Цветочек",
  " Машинка",
  " Пицца",
  " Инопланетянин"
  -- добавь свои темы!
}

local function clearBoard()
  print("Очищаем доску!")
  local pixels = {}
  for _, part in ipairs(workspace:GetDescendants()) do
    if part:IsA("BasePart") and part.Name == "Pixel" then
      table.insert(pixels, part)
    end
  end
  for _, pixel in ipairs(pixels) do
    pixel.BrickColor = BrickColor.new("Hot pink")
  end
  wait(0.3)
  for _, pixel in ipairs(pixels) do
    pixel.BrickColor = BrickColor.White()
  end
end

local function broadcastInfo(theme, timeLeft)
  for _, player in ipairs(Players:GetPlayers()) do
    gameInfoEvent:FireClient(player, theme, timeLeft)
  end
end

while true do
  local theme = themes[math.random(1, #themes)]
  for timer = 60, 0, -1 do
    broadcastInfo(theme, timer)
    wait(1)
  end
  clearBoard()
  wait(1)
end

📄 LocalScript 1 — цвет и рисование
local Players = game:GetService("Players")
local player = Players.LocalPlayer
local mouse = player:GetMouse()
local replicatedStorage = game:GetService("ReplicatedStorage")
local changeColorEvent = replicatedStorage:FindFirstChild("ChangeColorEvent")
local currentColor = BrickColor.Red()

changeColorEvent.OnClientEvent:Connect(function(newColor)
  currentColor = newColor
  print("Текущий цвет:", newColor.Name)
end)

local function onPixelClick(pixel)
  if pixel.BrickColor ~= currentColor then
    pixel.BrickColor = currentColor
  end
end

mouse.Button1Down:Connect(function()
  local target = mouse.Target
  if target and target.Name == "Pixel" then
    onPixelClick(target)
  end
end)
📄 LocalScript 2 — тема и таймер
local replicatedStorage = game:GetService("ReplicatedStorage")
local gameInfoEvent = replicatedStorage:FindFirstChild("GameInfoEvent")
local player = game.Players.LocalPlayer
local gui = player:WaitForChild("PlayerGui"):WaitForChild("DrawingGUI")
local themeLabel = gui:FindFirstChild("ThemeLabel")
local timerLabel = gui:FindFirstChild("TimerLabel")

gameInfoEvent.OnClientEvent:Connect(function(theme, timeLeft)
  if themeLabel then
    themeLabel.Text = "Тема: " .. theme
  end
  if timerLabel then
    timerLabel.Text = tostring(timeLeft)
  end
end)
