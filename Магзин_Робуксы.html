🛒 Магазин Робуксы

GUI магазина + Game Pass + выдача CocaCola в рюкзак

🥤
Миссия проекта: сделать магазин с кнопкой «Купить». Игрок открывает окно, жмёт Buy → покупает Game Pass за робаксы → получает инструмент CocaCola в Backpack. Нужны GUI, публикация игры и помощь учителя.

💡 На этом проекте ты:

  • соберёшь GUI: ShopButton, ShopFrame, Buy 🖥️
  • подключишь картинку товара через AssetId 🖼️
  • создашь Game Pass на сайте Roblox 🎟️
  • выдашь Tool из ReplicatedStorage после покупки 🥤

🎟️ Фаза 3 — Game Pass (с учителем) 20 мин

🎓 На этом этапе зови учителя: игру нужно опубликовать и сделать публичной.

-- LocalScript в ScreenGui
local MarketplaceService = game:GetService("MarketplaceService")
local buyButton = script.Parent.ShopFrame.Buy
local player = game.Players.LocalPlayer
local gamepassId = 1841593670 -- ← свой Pass ID!

buyButton.MouseButton1Click:Connect(function()
    local hasPass = false
    local success, errormessage = pcall(function()
        hasPass = MarketplaceService:UserOwnsGamePassAsync(player.UserId, gamepassId)
    end)

    if hasPass then
        print("У игрока уже есть кока-кола")
    else
        MarketplaceService:PromptGamePassPurchase(player, gamepassId)
    end
end)
-- Script в ServerScriptService
local MarketplaceService = game:GetService("MarketplaceService")
local Players = game:GetService("Players")
local gamepassId = 1841593670 -- ← тот же Pass ID!

local function giveCola(player)
    local cola = game.ReplicatedStorage.CocaCola:Clone()
    cola.Parent = player.Backpack
end

Players.PlayerAdded:Connect(function(player)
    local success, err = pcall(function()
        local hasPass = MarketplaceService:UserOwnsGamePassAsync(player.UserId, gamepassId)
        if hasPass then
            print("У игрока есть пасс, добавляем колу")
            giveCola(player)
        else
            print("У игрока нет пасса")
        end
    end)
    if not success then
        warn("Ошибка проверки пасса: " .. tostring(err))
    end
end)

local function onPromptGamePassPurchaseFinished(player, purchasedPassId, purchaseSuccess)
    if purchaseSuccess and purchasedPassId == gamepassId then
        print(player.Name .. " купил кока-колу")
        giveCola(player)
    else
        print(player.Name .. " не купил кока-колу")
    end
end

MarketplaceService.PromptGamePassPurchaseFinished:Connect(onPromptGamePassPurchaseFinished)
