☀️ Взрыв Солнца

Пляж, 10 солнышек, нарастающая яркость и финальный взрыв

💥
Миссия проекта: собрать пляжную локацию и разложить 10 SunToken. Каждый сбор делает мир ярче. Когда собраны все — солнце «взрывается», мир рушится, игроку показывают победу.

💡 На этом проекте ты:

  • оформишь пляж моделями из Toolbox 🏖️
  • создашь светящиеся SunToken (Neon + PointLight) 💛
  • изменишь Lighting по мере сбора 🌤️
  • запустишь финальный Explosion и Kick 💥

📚 Теория

1. Цикл катастрофы

① Сбор

Touched → уничтожить токен → счётчик +1

② Яркость

Lighting.Brightness растёт с каждым солнышком

③ Взрыв

При 10/10 — Explosion, очистка мира, Kick

2. Имя SunToken

Скрипт ищет все объекты с именем SunToken через GetDescendants(). Все 10 копий должны называться одинаково — не SunToken1, а именно SunToken.

3. Lighting

Сервис освещения мира. Меняя Brightness, ColorShift_Top, FogEnd, мы «нагреваем» пляж без новых моделей.

📜 Фаза 3 — Скрипт в ServerScriptService 20 мин

  1. ServerScriptService → добавь Script.
  2. Вставь код ниже.
  3. Play: собери токены — яркость растёт; на 10-м — взрыв и Kick.

Один Script в ServerScriptService управляет всей игрой.

-- Пляжный симулятор – сбор солнышек
local Lighting = game:GetService("Lighting")
local Players = game:GetService("Players")
local workspace = game:GetService("Workspace")

local NEED = 10
local collectedCount = 0

local countTokens = 0
for _, obj in ipairs(workspace:GetDescendants()) do
    if obj.Name == "SunToken" and obj:IsA("BasePart") then
        countTokens += 1
        print("Найдено солнышко:", obj:GetFullName())
    end
end
print("Всего солнышек на старте:", countTokens)
if countTokens < 10 then
    warn("Не хватает солнышек! Нужно 10, а есть " .. countTokens)
end

local function increaseBrightness(currentCount)
    local brightness = 1 + (currentCount / NEED) * 3
    if brightness > 4 then brightness = 4 end
    Lighting.Brightness = brightness
    local intensity = currentCount / NEED
    Lighting.ColorShift_Top = Color3.new(1, 0.8 + intensity * 0.2, 0.5 + intensity * 0.2)
    Lighting.FogEnd = 400 + (currentCount * 20)
    print("Яркость:", brightness)
end

local function explodeWorld()
    print("Собрано 10! Взрываем мир!")
    local explosion = Instance.new("Explosion")
    explosion.BlastRadius = 100
    explosion.BlastPressure = 500000
    explosion.Position = Vector3.new(0, 10, 0)
    explosion.Parent = workspace

    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart")
            and obj.Name ~= "SpawnLocation"
            and not obj.Parent:IsA("Model") then
            -- не трогаем части персонажа
            local hum = obj.Parent and obj.Parent:FindFirstChild("Humanoid")
            if not hum then
                obj:Destroy()
            end
        end
    end

    for _, player in ipairs(Players:GetPlayers()) do
        local msg = Instance.new("Message")
        msg.Text = "СОЛНЦЕ ВЗОРВАЛОСЬ! ПОБЕДА!"
        msg.Parent = player.PlayerGui
        task.wait(3)
        msg:Destroy()
    end

    task.wait(2)
    for _, player in ipairs(Players:GetPlayers()) do
        player:Kick("Ты собрал все солнышки и устроил взрыв! Игра окончена.")
    end
end

local function onTokenCollected(token)
    if not token or token.Parent == nil then return end
    token:Destroy()
    collectedCount += 1
    print("Собрано:", collectedCount, "/", NEED)
    increaseBrightness(collectedCount)
    if collectedCount >= NEED then
        explodeWorld()
    end
end

local function setupToken(token)
    if token:IsA("BasePart") and token.Name == "SunToken" then
        if token:GetAttribute("TouchedConnected") then return end
        token:SetAttribute("TouchedConnected", true)
        local connection
        connection = token.Touched:Connect(function(hit)
            local player = Players:GetPlayerFromCharacter(hit.Parent)
            if player and token.Parent ~= nil then
                connection:Disconnect()
                onTokenCollected(token)
            end
        end)
    end
end

for _, token in ipairs(workspace:GetDescendants()) do
    setupToken(token)
end
workspace.DescendantAdded:Connect(setupToken)
increaseBrightness(0)
✅
Атрибут TouchedConnected и Disconnect не дают собрать одно солнышко дважды.
⚠️
Kick выкинет из Play — это нормально для финала. В Studio после Kick нажми Stop.
